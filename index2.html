🟢 Integrasi Google Sheet Absensi

Sheet Sheet1 (log absensi): TANGGALABSEN | NIS | NAMA | KELAS | STATUSABSEN | TANGGAL

🔄 Sudah pernah pasang kode ini sebelumnya? Versi terbaru memperbaiki bug "data tidak tampil di Rekap/Laporan" akibat Google Sheets otomatis mengubah kolom TANGGALABSEN jadi tipe Date. Buka Apps Script Anda, timpa ulang seluruh kode lama dengan kode di bawah ini, lalu buat Deploy > Manage deployments > Edit (ikon pensil) > Version: New version > Deploy (bukan New deployment, supaya URL-nya tetap sama). Tidak perlu mengubah data yang sudah ada di sheet — kode baru otomatis membaca data lama dengan benar.

📋 Langkah Menghubungkan Google Sheet:

  1. Buka Google Sheet Anda (Link Google Sheet).
  2. Klik menu Extensions (Ekstensi) > Apps Script.
  3. Hapus semua kode yang ada, lalu salin (paste) kode berikut (sheet DATASISWA akan dibuat otomatis, tidak perlu dibuat manual):
var SHEET_ABSENSI = 'Sheet1';
var HEADERS_ABSENSI = ['TANGGALABSEN', 'NIS', 'NAMA', 'KELAS', 'STATUSABSEN', 'TANGGAL'];

var SHEET_SISWA = 'DATASISWA';
var HEADERS_SISWA = ['NIS', 'NAMA', 'KELAS'];

function getOrCreateSheet_(name, headers) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold');
  }
  // PENTING: paksa kolom TANGGALABSEN & NIS berformat TEKS ('@'), supaya Google Sheets
  // TIDAK otomatis mengubahnya jadi tipe Date/Number. Kalau dibiarkan default, Sheets
  // sering mengonversi string "2026-07-21" jadi objek Date, sehingga perbandingan
  // rentang tanggal (string vs Date) di script ini gagal cocok walau datanya ada.
  if (name === SHEET_ABSENSI) {
    sheet.getRange('A:A').setNumberFormat('@'); // TANGGALABSEN
    sheet.getRange('B:B').setNumberFormat('@'); // NIS
    sheet.getRange('F:F').setNumberFormat('@'); // TANGGAL (timestamp simpan)
  } else if (name === SHEET_SISWA) {
    sheet.getRange('A:A').setNumberFormat('@'); // NIS
  }
  return sheet;
}

/**
 * Menormalkan nilai tanggal dari sel Sheet menjadi string 'yyyy-MM-dd',
 * baik sel itu tersimpan sebagai teks maupun (kalau terlanjur) sebagai objek Date.
 * Dipakai di semua tempat yang membandingkan/menyimpan TANGGALABSEN supaya
 * konsisten apa pun format aslinya di sheet.
 */
function normalizeTanggal_(val) {
  if (val instanceof Date) {
    return Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  }
  return String(val || '').trim();
}

function jsonOutput_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var action = (e.parameter && e.parameter.action) || 'ping';
  if (action === 'getSiswa') {
    var sheet = getOrCreateSheet_(SHEET_SISWA, HEADERS_SISWA);
    var lastRow = sheet.getLastRow();
    var data = [];
    if (lastRow >= 2) {
      var values = sheet.getRange(2, 1, lastRow - 1, 3).getValues();
      data = values.filter(function(r){ return r[0] !== '' && r[1] !== ''; })
                    .map(function(r){ return {nis:String(r[0]).trim(), nama:String(r[1]).trim(), kelas:String(r[2]).trim()}; });
    }
    return jsonOutput_({ result: 'success', data: data });
  }
  if (action === 'getAbsensi') {
    var sheetLog = getOrCreateSheet_(SHEET_ABSENSI, HEADERS_ABSENSI);
    var lastRowLog = sheetLog.getLastRow();
    var kelasParam = String(e.parameter.kelas || '').toUpperCase().trim();
    var mulaiParam = e.parameter.mulai || '';
    var selesaiParam = e.parameter.selesai || '';
    var dataLog = [];
    if (lastRowLog >= 2) {
      var valuesLog = sheetLog.getRange(2, 1, lastRowLog - 1, HEADERS_ABSENSI.length).getValues();
      valuesLog.forEach(function(r){
        var tglAbsen = normalizeTanggal_(r[0]);
        var kelasRow = String(r[3]).toUpperCase().trim();
        if (kelasParam && kelasRow !== kelasParam) return;
        if (mulaiParam && tglAbsen < mulaiParam) return;
        if (selesaiParam && tglAbsen > selesaiParam) return;
        dataLog.push({ tanggal: tglAbsen, nis: String(r[1]).trim(), nama: String(r[2]), kelas: String(r[3]), status: String(r[4]) });
      });
    }
    return jsonOutput_({ result: 'success', data: dataLog });
  }
  return jsonOutput_({ result: 'ok', message: 'Apps Script Absensi aktif.' });
}

function doPost(e) {
  try {
    var body = JSON.parse(e.postData.contents);
    if (body.action === 'saveAbsensi') return simpanAbsensi_(body.payload);
    if (body.action === 'importSiswa') return importSiswa_(body.payload);
    return jsonOutput_({ result: 'error', error: 'Action tidak dikenali.' });
  } catch (err) {
    return jsonOutput_({ result: 'error', error: err.toString() });
  }
}

function simpanAbsensi_(data) {
  // UPSERT: kalau sudah ada baris dengan TANGGALABSEN+NIS yang sama, baris itu DIPERBARUI
  // (bukan ditambah baris baru), supaya Simpan berkali-kali di tanggal yang sama tidak
  // menumpuk data duplikat di sheet.
  var sheet = getOrCreateSheet_(SHEET_ABSENSI, HEADERS_ABSENSI);
  var lastRow = sheet.getLastRow();
  var existing = lastRow >= 2 ? sheet.getRange(2, 1, lastRow - 1, HEADERS_ABSENSI.length).getValues() : [];

  var indexMap = {};
  existing.forEach(function(r, idx){
    var key = normalizeTanggal_(r[0]) + '||' + String(r[1]).trim();
    indexMap[key] = idx;
  });

  data.forEach(function(it){
    var key = normalizeTanggal_(it.TANGGALABSEN) + '||' + String(it.NIS).trim();
    var rowValues = [it.TANGGALABSEN, it.NIS, it.NAMA, it.KELAS, it.STATUS_ABSEN, it.TANGGAL];
    if (indexMap.hasOwnProperty(key)) {
      existing[indexMap[key]] = rowValues;
    } else {
      existing.push(rowValues);
      indexMap[key] = existing.length - 1;
    }
  });

  if (lastRow >= 2) sheet.getRange(2, 1, lastRow - 1, HEADERS_ABSENSI.length).clearContent();
  if (existing.length > 0) sheet.getRange(2, 1, existing.length, HEADERS_ABSENSI.length).setValues(existing);

  return jsonOutput_({ result: 'success', saved: data.length, total: existing.length });
}

function importSiswa_(data) {
  var sheet = getOrCreateSheet_(SHEET_SISWA, HEADERS_SISWA);
  var kelasSet = {};
  data.forEach(function(s){ kelasSet[String(s.KELAS).toUpperCase().trim()] = true; });

  var lastRow = sheet.getLastRow();
  var existing = lastRow >= 2 ? sheet.getRange(2, 1, lastRow - 1, 3).getValues() : [];
  var kept = existing.filter(function(r){ return !kelasSet[String(r[2]).toUpperCase().trim()]; });
  var baru = data.map(function(s){ return [String(s.NIS).trim(), String(s.NAMA).trim(), String(s.KELAS).toUpperCase().trim()]; });
  var final = kept.concat(baru);

  if (lastRow >= 2) sheet.getRange(2, 1, lastRow - 1, 3).clearContent();
  if (final.length > 0) sheet.getRange(2, 1, final.length, 3).setValues(final);

  return jsonOutput_({ result: 'success', imported: baru.length, total: final.length });
}

📥 Import Data Siswa (CSV)

Simpan Excel Anda ke format CSV (Comma Delimited). Struktur kolom: NIS, NAMA, KELAS (contoh: 89995905, AIRIN RACHMI SEPTIANA, XII TKJ 3). Jika file punya kolom KELAS, kelas tiap siswa otomatis terbaca per-baris — bisa impor banyak kelas sekaligus dalam satu file.

Dipakai sebagai fallback jika file CSV tidak menyertakan kolom KELAS.
